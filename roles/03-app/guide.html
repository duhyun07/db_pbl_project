1프로젝트 구조

① 이 단계에서 만드는 것

app/ 폴더와 app/pages/ 폴더, 그리고 아래 구조의 빈 파일들.

② 왜 이렇게 하나

3app/db.py — DB 연결

① 이 단계에서 만드는 것

app/db.py — 연결 함수 get_connection(), 조회 함수 fetch_df()(결과를 DataFrame으로), 실행 함수 execute()(commit/rollback).

② 왜 이렇게 하나

모든 SQL은 cursor.execute(sql, params)처럼 값을 따로 넘기는 방식(파라미터 바인딩)으로만 실행해요. "WHERE cve_id = '" + 입력값 + "'"처럼 이어 붙이면, 입력값에 ' OR 1=1 --를 넣는 것만으로 조건이 무력화돼 모든 행이 보이거나 데이터가 바뀌어요(SQL Injection). 바인딩을 쓰면 PyMySQL이 값을 따옴표로 감싸 "그냥 문자열"로만 넣어요. 보안학과 발표라 꼭 질문이 나오는 부분이에요.

③ 코드

"""DB 연결과 쿼리 실행을 한곳에 모은 모듈. 화면 코드는 이 파일의 함수만 쓴다."""
import os
from pathlib import Path

import pandas as pd
import pymysql
from dotenv import load_dotenv

# 레포 루트의 .env를 읽는다 (app/db.py 기준 한 단계 위 폴더)
load_dotenv(Path(__file__).resolve().parent.parent / ".env")


def get_connection():
    """새 DB 연결을 만든다. autocommit=False라서 변경은 commit()해야 반영된다."""
    return pymysql.connect(
        host=os.getenv("MYSQL_HOST", "127.0.0.1"),
        port=int(os.getenv("MYSQL_PORT", "3306")),
        user=os.getenv("MYSQL_USER"),
        password=os.getenv("MYSQL_PASSWORD"),
        database=os.getenv("MYSQL_DATABASE", "vuln_mgmt"),
        charset="utf8mb4",
        cursorclass=pymysql.cursors.DictCursor,   # 결과 행을 {"컬럼명": 값} 형태로 받는다
        autocommit=False,
    )


def fetch_df(sql, params=None):
    """SELECT 결과를 pandas DataFrame으로 돌려준다.

    params에는 %s 자리에 들어갈 값을 튜플이나 리스트로 넘긴다.
    값은 PyMySQL이 따옴표 처리(escape)를 해서 넣어 주므로 SQL Injection이 생기지 않는다.
    """
    conn = get_connection()
    try:
        with conn.cursor() as cur:
            cur.execute(sql, params)
            rows = cur.fetchall()
            columns = [c[0] for c in cur.description]
        return pd.DataFrame(rows, columns=columns)
    finally:
        conn.close()


def execute(sql, params=None):
    """INSERT/UPDATE/DELETE 한 문장을 실행하고 commit한다. 영향받은 행 수를 돌려준다."""
    conn = get_connection()
    try:
        with conn.cursor() as cur:
            affected = cur.execute(sql, params)
        conn.commit()
        return affected
    except Exception:
        conn.rollback()   # 실패하면 이 연결에서 한 변경을 모두 취소
        raise
    finally:
        conn.close()

4app/queries.py — catalog 실행

① 이 단계에서 만드는 것

app/queries.py — ②의 sql/query_catalog.json을 읽고 run_query("id", {파라미터})로 실행하는 함수. catalog가 오기 전에 쓸 임시 쿼리 2개 포함.

② 왜 이렇게 하나

분석 SQL을 앱에서 새로 쓰지 않고 ②가 검증한 쿼리를 그대로 써야 "SQL 결과 = 화면 숫자"가 보장돼요. catalog의 type대로 값을 변환해서 넘기니, 화면 위젯에서 이상한 값이 와도 SQL 문장 자체는 바뀌지 않아요.

③ 코드

"""②가 만든 sql/query_catalog.json을 읽어서 id로 쿼리를 실행한다."""
import json
from pathlib import Path

from db import fetch_df

CATALOG_PATH = Path(__file__).resolve().parent.parent / "sql" / "query_catalog.json"

# [임시] catalog가 아직 없을 때만 쓰는 쿼리. ②의 query_catalog.json이 오면 이 부분은 지운다.
TEMP_QUERIES = [
    {
        "id": "temp_open_by_server",
        "title": "[임시] 서버별 미해결 건수",
        "description": "catalog가 오기 전까지 화면을 만들어 보기 위한 임시 쿼리",
        "sql": "SELECT hostname, COUNT(*) AS open_count FROM v_open_risks GROUP BY hostname ORDER BY open_count DESC",
        "params": [],
    },
    {
        "id": "temp_high_cvss",
        "title": "[임시] CVSS 기준 이상 미해결 건",
        "description": "catalog가 오기 전까지 쓰는 임시 쿼리",
        "sql": "SELECT hostname, cve_id, cvss_score, assignee_name FROM v_open_risks "
               "WHERE cvss_score >= %(min_cvss)s ORDER BY cvss_score DESC",
        "params": [{"name": "min_cvss", "label": "최소 CVSS", "type": "float", "default": 7.0}],
    },
]


def load_catalog():
    """catalog 파일이 있으면 그것을, 없으면 임시 쿼리를 돌려준다."""
    if CATALOG_PATH.exists():
        with open(CATALOG_PATH, encoding="utf-8") as f:
            return json.load(f)["queries"]
    return TEMP_QUERIES


def get_query(query_id):
    for q in load_catalog():
        if q["id"] == query_id:
            return q
    raise KeyError(f"catalog에 없는 쿼리 id: {query_id}")


def run_query(query_id, values=None):
    """id로 쿼리를 찾아 실행한다. values에 없는 파라미터는 catalog의 default를 쓴다."""
    q = get_query(query_id)
    values = values or {}
    params = {}
    for p in q["params"]:
        raw = values.get(p["name"], p["default"])
        # catalog에 적힌 type대로 변환해서 넘긴다 (화면 위젯 값이 문자열로 와도 안전하게)
        params[p["name"]] = {"int": int, "float": float, "str": str, "date": str}[p["type"]](raw)
    return fetch_df(q["sql"], params or None)

② 왜 이렇게 하나

③ 코드

"""대시보드 (첫 화면). 실행: streamlit run app/main.py"""
import streamlit as st

from db import fetch_df

st.set_page_config(page_title="취약점·패치 관리", layout="wide")
st.title("취약점·패치 관리 대시보드")


@st.cache_data(ttl=60)   # 60초 동안 같은 결과를 재사용 → 화면을 움직일 때마다 DB에 다시 묻지 않는다
def load_summary():
    return fetch_df("""
        SELECT
          (SELECT COUNT(*) FROM systems)                                 AS server_count,
          (SELECT COUNT(DISTINCT cve_id) FROM system_vulnerabilities)    AS cve_count,
          (SELECT COUNT(*) FROM v_open_risks)                            AS open_count,
          (SELECT COUNT(*) FROM v_open_risks WHERE cvss_score >= 7.0)    AS high_open_count,
          (SELECT COUNT(*) FROM v_open_risks WHERE is_overdue = 1)       AS overdue_count
    """).iloc[0]


@st.cache_data(ttl=60)
def load_patch_rate():
    return fetch_df("SELECT hostname, environment, total_count, patched_count, unresolved_count, "
                    "risk_accepted_count, patch_rate_pct FROM v_system_patch_rate ORDER BY patch_rate_pct")


summary = load_summary()
c1, c2, c3, c4, c5 = st.columns(5)
c1.metric("서버", int(summary["server_count"]))
c2.metric("취약점 (서로 다른 CVE)", int(summary["cve_count"]))
c3.metric("미해결 조치 건", int(summary["open_count"]))
c4.metric("고위험 미해결 (CVSS 7.0+)", int(summary["high_open_count"]))
c5.metric("기한 초과", int(summary["overdue_count"]))

st.subheader("서버별 패치 완료율")
rate = load_patch_rate()
st.dataframe(rate, hide_index=True, width="stretch")
st.bar_chart(rate.set_index("hostname")["patch_rate_pct"].astype(float))   # DECIMAL → float (차트용)
st.caption("완료율 = patched / 전체 조치 건 × 100. 조치 건이 없는 서버는 빈칸(NULL)으로 나와요.")

6검색 화면 (app/pages/1_search.py)

① 이 단계에서 만드는 것

필터 위젯(CVE 번호, 소프트웨어, CVSS 범위, 조치 상태) → 파라미터 → 쿼리 → st.dataframe.

② 왜 이렇게 하나

필터에 따라 WHERE 조건 개수가 바뀌니 SQL을 조립해야 해요. 이때 코드에 고정된 조건 문장만 붙이고, 사용자가 입력한 값은 전부 params 리스트로 넘겨요. 이게 "조립"과 "이어 붙이기"의 차이예요. 실행한 SQL을 화면에 펼쳐 보여주면 발표 때 "화면 뒤에서 이 SQL이 돈다"를 바로 보여줄 수 있어요.

③ 코드

"""검색 화면: 필터 위젯 → 파라미터 → 쿼리 → 표"""
import streamlit as st

from db import fetch_df

st.title("취약점 검색")

col1, col2, col3, col4 = st.columns(4)
cve_text = col1.text_input("CVE 번호 (일부만 입력해도 됨)")
software = col2.selectbox("소프트웨어", ["전체", "Apache HTTP Server", "OpenSSH"])
cvss_range = col3.slider("CVSS 범위", 0.0, 10.0, (0.0, 10.0), step=0.1)
statuses = col4.multiselect("조치 상태", ["open", "in_progress", "patched", "risk_accepted"],
                            default=["open", "in_progress"])
include_no_score = st.checkbox("CVSS 점수가 없는 취약점도 포함", value=True)

# 조건 문장은 코드에 고정된 문자열만 붙이고, 사용자가 입력한 값은 전부 params로 넘긴다.
conditions = []
params = []
if cve_text:
    conditions.append("v.cve_id LIKE %s")
    params.append(f"%{cve_text}%")
if software != "전체":
    conditions.append("sw.name = %s")
    params.append(software)
if include_no_score:
    conditions.append("(v.cvss_score BETWEEN %s AND %s OR v.cvss_score IS NULL)")
else:
    conditions.append("v.cvss_score BETWEEN %s AND %s")
params.extend(cvss_range)
if statuses:
    conditions.append("sv.status IN (" + ", ".join(["%s"] * len(statuses)) + ")")
    params.extend(statuses)

sql = """
SELECT sv.sv_id, s.hostname, sw.name AS software, v.cve_id, v.cvss_score,
       sv.status, st.name AS assignee, sv.due_date
FROM system_vulnerabilities sv
JOIN systems s          ON s.system_id = sv.system_id
JOIN software sw        ON sw.software_id = sv.software_id
JOIN vulnerabilities v  ON v.cve_id = sv.cve_id
LEFT JOIN staff st      ON st.staff_id = sv.assignee_id
WHERE """ + " AND ".join(conditions) + """
ORDER BY v.cvss_score DESC, s.hostname"""

result = fetch_df(sql, params)
st.write(f"검색 결과 {len(result)}건")
st.dataframe(result, hide_index=True, width="stretch")
with st.expander("실행한 SQL 보기"):
    st.code(sql, language="sql")
    st.write("파라미터:", params)

7패치 관리 화면 (app/pages/2_patch.py)

① 이 단계에서 만드는 것

조치 건 선택 → 상태·담당자 변경 폼 → UPDATE … updated_by → commit → 트리거가 남긴 이력 확인.

② 왜 이렇게 하나

  • updated_by를 같은 UPDATE에: 이력은 ①의 트리거가 patch_history에 자동으로 남기고, 변경자는 NEW.updated_by 값을 써요. 빠뜨리면 이전 변경자의 이름으로 기록돼요 (docs/db_dictionary.md 맨 위 규칙).
  • 한 트랜잭션: 상태 변경과 비고 기록이 둘 다 되거나 둘 다 안 되게 해요. 중간에 실패하면 rollback으로 원래대로 돌아가요.
  • FOR UPDATE: 두 사람이 같은 건을 동시에 바꿔도 하나씩 처리되게 행을 잠가요.
  • patched_at: patched로 바꿀 때 완료 시각이 없으면 CHECK 제약(3819)에 걸리니 SQL에서 NOW()로 채워요.

③ 코드

"""패치 관리 화면: 목록 선택 → 상태·담당자 변경 → 한 트랜잭션으로 저장 → 이력 확인"""
import pandas as pd
import pymysql
import streamlit as st

from db import fetch_df, get_connection

st.title("패치 관리")

STATUSES = ["open", "in_progress", "patched", "risk_accepted"]


@st.cache_data(ttl=60)
def load_items():
    return fetch_df("""
        SELECT sv.sv_id, s.hostname, sv.cve_id, sv.status, sv.assignee_id, sv.due_date
        FROM system_vulnerabilities sv
        JOIN systems s ON s.system_id = sv.system_id
        ORDER BY sv.sv_id""")


@st.cache_data(ttl=300)
def load_staff():
    return fetch_df("SELECT staff_id, name FROM staff ORDER BY staff_id")


def change_status(sv_id, new_status, assignee_id, staff_id, note):
    """상태 변경과 비고 기록을 하나의 트랜잭션으로 처리한다. 둘 중 하나라도 실패하면 둘 다 취소."""
    conn = get_connection()
    try:
        with conn.cursor() as cur:
            # 0) 현재 상태를 읽으면서 행을 잠근다 (FOR UPDATE: 커밋 전까지 다른 사람이 이 행을 못 바꿈)
            cur.execute("SELECT status FROM system_vulnerabilities WHERE sv_id = %s FOR UPDATE", (sv_id,))
            old_status = cur.fetchone()["status"]

            # 1) 상태 변경. updated_by를 반드시 같이 SET → 트리거가 patch_history.changed_by에 이 값을 기록
            cur.execute(
                """UPDATE system_vulnerabilities
                      SET status      = %s,
                          patched_at  = IF(%s = 'patched', COALESCE(patched_at, NOW()), patched_at),
                          assignee_id = %s,
                          updated_by  = %s
                    WHERE sv_id = %s""",
                (new_status, new_status, assignee_id, staff_id, sv_id),
            )
            # 2) 트리거가 방금 만든 이력 행에 비고를 남긴다 (상태가 실제로 바뀌었을 때만 이력 행이 생김)
            if note and old_status != new_status:
                cur.execute("SELECT MAX(history_id) AS hid FROM patch_history WHERE sv_id = %s", (sv_id,))
                hid = cur.fetchone()["hid"]
                cur.execute("UPDATE patch_history SET note = %s WHERE history_id = %s", (note, hid))
        conn.commit()
    except Exception:
        conn.rollback()
        raise
    finally:
        conn.close()


items = load_items()
staff = load_staff()
staff_names = dict(zip(staff["staff_id"], staff["name"]))

# 저장 직후 st.rerun()으로 다시 그려지므로, 성공 메시지는 session_state에 넣어 두었다가 보여준다
if "message" in st.session_state:
    st.success(st.session_state.pop("message"))

st.dataframe(items, hide_index=True, width="stretch")

# 조치 건 선택은 폼 밖에 둔다. 폼 안의 위젯은 "저장"을 누르기 전까지 값이 반영되지 않아서
# 아래의 "현재 상태" 표시와 기본값이 바뀌지 않기 때문이다.
sv_id = st.selectbox("조치 건 (sv_id)", items["sv_id"].tolist())
current = items[items["sv_id"] == sv_id].iloc[0]
st.caption(f"현재: {current['hostname']} / {current['cve_id']} / {current['status']}")

with st.form("change_form"):
    new_status = st.selectbox("새 상태", STATUSES, index=STATUSES.index(current["status"]))
    assignee_options = [None] + staff["staff_id"].tolist()
    current_assignee = None if pd.isna(current["assignee_id"]) else int(current["assignee_id"])
    assignee_id = st.selectbox("담당자", assignee_options, index=assignee_options.index(current_assignee),
                               format_func=lambda x: "(미배정)" if x is None else staff_names[x])
    staff_id = st.selectbox("변경하는 사람 (updated_by)", staff["staff_id"].tolist(),
                            format_func=lambda x: staff_names[x])
    note = st.text_input("비고 (위험 수용 사유 등)")
    submitted = st.form_submit_button("저장")

if submitted:
    try:
        change_status(int(sv_id), new_status, assignee_id, int(staff_id), note)
        st.cache_data.clear()   # 캐시를 지워야 표에 바뀐 값이 바로 보인다
        st.session_state["message"] = "저장했어요."
        st.rerun()              # 바뀐 값으로 화면 전체를 다시 그린다
    except pymysql.err.OperationalError as e:
        if e.args[0] == 3819:
            st.error("CHECK 제약 위반: patched로 바꿀 때는 완료 시각이 필요해요.")
        else:
            st.error(f"DB 오류: {e}")
    except pymysql.err.IntegrityError as e:
        st.error(f"무결성 오류: {e}")

st.subheader("변경 이력")
history = fetch_df("""
    SELECT h.history_id, h.old_status, h.new_status, st.name AS changed_by, h.changed_at, h.note
    FROM patch_history h
    LEFT JOIN staff st ON st.staff_id = h.changed_by
    WHERE h.sv_id = %s
    ORDER BY h.history_id""", (int(sv_id),))
st.dataframe(history, hide_index=True, width="stretch")

"""분석 화면: ②의 query_catalog.json 쿼리를 골라 실행하고 표·차트로 보여준다."""
import streamlit as st

from queries import load_catalog, run_query

st.title("분석")

catalog = load_catalog()
titles = {q["title"]: q for q in catalog}
chosen = titles[st.selectbox("분석 질문", list(titles))]
st.caption(chosen["description"])

# catalog의 params 정의대로 입력 위젯을 자동으로 만든다
values = {}
for p in chosen["params"]:
    if p["type"] == "int":
        values[p["name"]] = st.number_input(p["label"], value=int(p["default"]), step=1)
    elif p["type"] == "float":
        values[p["name"]] = st.number_input(p["label"], value=float(p["default"]), step=0.1)
    else:
        values[p["name"]] = st.text_input(p["label"], value=str(p["default"]))

result = run_query(chosen["id"], values)
st.dataframe(result, hide_index=True, width="stretch")

# 숫자 컬럼이 있으면 첫 번째 컬럼 기준 막대 차트
numeric = result.select_dtypes("number").columns.tolist()
if len(result) and numeric:
    st.bar_chart(result.set_index(result.columns[0])[numeric[0]])

with st.expander("실행한 SQL 보기"):
    st.code(chosen["sql"], language="sql")
    st.write("파라미터:", values)

11(11/28 이후, 여유가 되면) LLM 구조

LLM은 부가 기능이에요. 1~10단계가 모두 끝나고 시연이 안정된 뒤에만 시작해요. 여기서는 구조만 설명하고 코드는 쓰지 않아요.

① 이 단계에서 만드는 것

자연어 질문을 받아 catalog 쿼리 하나를 고르게 하는 화면 (예: app/pages/4_ask.py).

② 왜 이렇게 하나

LLM이 SQL을 직접 쓰게 하면 틀린 SQL이나 위험한 SQL이 실행될 수 있어요. 대신 ②가 검증한 쿼리 중에서 고르게만 하면, 실행되는 SQL은 항상 검증된 것이고 LLM은 "어떤 질문인지 분류하는 일"만 해요.

③ 흐름

1. 사용자 질문         "기한 3일 넘은 미해결 건 보여줘"
2. LLM에게 전달        질문 + catalog의 id·title·description·params 목록
                       → "이 중 하나를 골라 JSON으로만 답해"
3. LLM 응답 (JSON)     {"id": "q5_overdue_open", "params": {"min_days_overdue": 3}}
4. 검증 (Python)       id가 catalog에 있는가? params 이름·타입이 맞는가? 값 범위가 정상인가?
                       → 하나라도 아니면 5로 가지 않고 드롭다운 화면으로
5. 실행               run_query(id, params)   (4단계 함수 그대로)
6. 요약               결과 행만 LLM에게 주고 "이 행들만 근거로 요약" → 화면에
                       실행한 SQL 원문과 결과 표를 요약 옆에 같이 표시
실패 시               3·4·5 중 어디서든 실패하면 8단계의 드롭다운으로 같은 쿼리를 고르게 함

④ 확인 방법

LLM 없이(API 키를 지운 상태로) 앱을 켜도 1~10단계 화면이 모두 정상이어야 해요. 그게 "LLM 없이도 시연이 된다"의 확인이에요.

⑤ 막히기 쉬운 부분

위험대응
LLM이 catalog에 없는 id를 지어냄4번 검증에서 거르고 드롭다운으로
요약에 결과에 없는 숫자가 섞임요약 옆에 결과 표를 항상 같이 보여주기
API 키가 커밋됨키는 .env에만. 키 이름만 .env.example에 추가

⑥ 다음 사람에게 넘길 것

catalog의 description이 분류 정확도를 좌우해요. 설명이 애매한 질문은 ②에게 문장 수정을 부탁해요.
